## 模块 16：钓鱼系统（Fishing System）

### 16.1 模块定位

武侠世界中的**休闲生活玩法**，提供战斗之外的放松体验。钓鱼不仅是收集鱼类，还关联食材（烹饪）、任务道具、收藏品、成就系统，是武侠游戏"江湖气息"的重要组成部分。

### 16.2 功能清单

| 分类 | 功能 | 优先级 | 说明 |
|------|------|--------|------|
| **钓具** | 鱼竿（多种品质/等级）、鱼饵（不同鱼偏好）、鱼线/鱼钩（影响成功率） | P0 | |
| **钓鱼地点** | 不同水域（河流/湖泊/海边/秘境/寒潭），不同地点鱼群不同 | P0 | 武侠场景特色 |

```gdscript
# data/runtime/fishing/caught_fish.gd —— 钓到的鱼实例
class_name CaughtFish
extends RefCounted

var fish_id: String = ""
var weight: float = 0.0           # 实际重量（斤）
var rarity: int = 0
var catch_time: int = 0
var catch_spot: String = ""
var is_biggest: bool = false      # 是否该鱼种最大记录
var quality: int = FishQuality.NORMAL  # 普通/精品/极品（影响价格和烹饪效果）

func get_data() -> FishData:
    return ConfigManager.get_fish(fish_id)

func get_price() -> int:
    var data: FishData = get_data()
    if data == null:
        return 0
    var price: float = data.base_price * weight
    match quality:
        FishQuality.FINE: price *= 1.5
        FishQuality.PREMIUM: price *= 2.5
    return int(price)

func serialize() -> Dictionary:
    return {
        "id": fish_id, "w": weight, "r": rarity,
        "t": catch_time, "spot": catch_spot, "q": quality,
    }

static func deserialize(data: Dictionary) -> CaughtFish:
    var f := CaughtFish.new()
    f.fish_id = data.get("id", "")
    f.weight = data.get("w", 0.0)
    f.rarity = data.get("r", 0)
    f.catch_time = data.get("t", 0)
    f.catch_spot = data.get("spot", "")
    f.quality = data.get("q", FishQuality.NORMAL)
    return f
```

### 16.4 核心业务逻辑

#### 16.4.1 钓鱼完整流程状态机

```
IDLE（空闲）
  ↓ 玩家交互钓鱼点 + 装备鱼竿
CASTING（抛竿）—— 播放抛竿动画，选择钓点（力度条小游戏）
  ↓ 抛竿完成
WAITING（等待咬钩）—— 等待时间 3~30秒，鱼漂微动提示
  ↓ 鱼咬钩（概率判定）
BITING（咬钩）—— 鱼漂下沉，玩家需在0.5~1.5秒内提竿
  ├─ 提竿及时 → FIGHTING（遛鱼）
  ├─ 提竿过早 → 鱼跑了，回到WAITING
  └─ 提竿过晚 → 鱼跑了，回到WAITING
FIGHTING（遛鱼）—— QTE张力控制，持续3~10秒
  ├─ 张力控制得当，鱼距离归零 → CAUGHT
  ├─ 张力过高（鱼线断）→ 鱼跑了，鱼竿耐久-
  ├─ 张力过低（鱼挣脱）→ 鱼跑了
  └─ 遛鱼超时 → 鱼跑了
CAUGHT（钓获）—— 展示鱼的信息，获得鱼，经验+，图鉴更新
  ↓ 自动或手动
IDLE
```

#### 16.4.2 咬钩概率计算

```gdscript
func _calculate_bite_chance(spot: FishingSpotData, fish: FishData, rod: FishingRodData, bait: BaitData) -> float:
    var chance: float = fish.base_catch_rate
    chance += rod.catch_rate_bonus
    if bait != null:
        if fish.preferred_bait.has(bait.id):
            chance *= bait.catch_rate_multiplier * 1.5
        else:
            chance *= bait.catch_rate_multiplier
        chance += bait.rare_bonus
    var level_diff: int = fishing_level - fish.min_fishing_level
    chance *= clamp(1.0 + level_diff * 0.05, 0.5, 2.0)
    var current_time: int = WeatherTimeService.get_time_of_day()
    if fish.time_conditions.size() > 0 and not fish.time_conditions.has(current_time):
        chance *= 0.3
    var current_weather: int = WeatherTimeService.get_weather()
    if fish.weather_conditions.size() > 0 and not fish.weather_conditions.has(current_weather):
        chance *= 0.5
    if rod.rod_tier < spot.required_rod_tier:
        chance *= 0.2
    return clamp(chance, 0.01, 0.95)
```

#### 16.4.3 鱼群选择（加权随机）

```gdscript
func _select_fish(spot: FishingSpotData) -> FishData:
    var valid_entries: Array[FishingSpotEntry] = []
    var total_weight: float = 0.0
    for entry in spot.fish_table:
        var fish: FishData = ConfigManager.get_fish(entry.fish_id)
        if fish == null:
            continue
        if fishing_level < fish.min_fishing_level:
            continue
        valid_entries.append(entry)
        total_weight += entry.weight
    if valid_entries.is_empty():
        return null
    var roll: float = randf() * total_weight
    var accumulated: float = 0.0
    for entry in valid_entries:
        accumulated += entry.weight
        if roll <= accumulated:
            return ConfigManager.get_fish(entry.fish_id)
    return ConfigManager.get_fish(valid_entries[-1].fish_id)
```

#### 16.4.4 遛鱼QTE逻辑

```gdscript
func _update_fighting(delta: float) -> void:
    if current_phase != FishingPhase.FIGHTING:
        return
    fight_timer -= delta
    var fish: FishData = current_fish
    var rod: FishingRodData = _get_equipped_rod_data()
    if randf() < 0.3 * delta * 10:
        _fish_direction = randf_range(-1.0, 1.0)
    var reeling: float = 1.0 if _is_reel_pressed else 0.0
    var tension_delta: float = (reeling * 0.8 + abs(_fish_direction) * fish.struggle_intensity * 0.3 - 0.2) * delta
    tension = clamp(tension + tension_delta, 0.0, 1.0)
    var distance_delta: float = (reeling * 0.15 - _fish_direction * fish.struggle_intensity * 0.1) * delta
    fish_distance = clamp(fish_distance + distance_delta, 0.0, 1.0)
    if tension >= 1.0:
        _line_break()
        return
    if fish_distance <= 0.0:
        _catch_fish()
        return
    if fight_timer <= 0:
        _fish_escape()
        return
```

#### 16.4.5 钓获处理

```gdscript
func _catch_fish() -> void:
    var fish: FishData = current_fish
    var weight: float = randf_range(fish.weight_min, fish.weight_max)
    var quality_roll: float = randf() + GameServices.player.luck * 0.01
    var quality: int = FishQuality.NORMAL
    if quality_roll > 0.95:
        quality = FishQuality.PREMIUM
    elif quality_roll > 0.75:
        quality = FishQuality.FINE
    var caught := CaughtFish.new()
    caught.fish_id = fish.id
    caught.weight = weight
    caught.rarity = fish.rarity
    caught.catch_time = Time.get_unix_time_from_system()
    caught.catch_spot = current_spot
    caught.quality = quality
    if fish_basket.size() >= basket_capacity:
        EventBus.notification_show.emit("鱼篓已满，%s逃走了" % tr(fish.name_key))
        _reset_to_idle()
        return
    fish_basket.append(caught)
    total_catches += 1
    total_weight_caught += weight
    _update_fish_dex(fish, caught)
    var exp_gain: int = int(10 * (1 + fish.rarity * 5) * (1 + weight / fish.weight_max))
    _gain_fishing_exp(exp_gain)
    if biggest_fish == null or weight > biggest_fish.weight:
        biggest_fish = caught
        caught.is_biggest = true
        EventBus.notification_show.emit("新纪录！最大的%s：%.1f斤" % [tr(fish.name_key), weight])
    if fish.rarity > rarest_fish_caught:
        rarest_fish_caught = fish.rarity
    if fish.rarity >= FishRarity.LEGENDARY:
        _trigger_special_event(fish, caught)
    _consume_rod_durability(1.0)
    current_phase = FishingPhase.CAUGHT
    EventBus.fishing_caught.emit(caught)
    EventBus.achievement_check.emit("fish_caught", fish.id)
```

#### 16.4.6 钓鱼经验与等级

```gdscript
func _gain_fishing_exp(amount: int) -> void:
    fishing_exp += amount
    while fishing_exp >= exp_to_next and fishing_level < 50:
        fishing_exp -= exp_to_next
        fishing_level += 1
        exp_to_next = int(exp_to_next * 1.2 + 50)
        EventBus.fishing_level_up.emit(fishing_level)
        EventBus.notification_show.emit("钓鱼等级提升至%d级" % fishing_level)
        _check_new_fish_unlock()
```

### 17.4 核心业务逻辑

#### 17.4.1 打造完整流程

```
玩家交互打造台 → 打开打造界面
    ↓
选择配方 → 显示材料需求/成品预览/成功率
    ↓
选择材料（从背包中选择，可选材料可替换）
    ↓ 点击打造
检查：技能等级/打造台/材料/燃料
    ↓ 全部满足
消耗材料+燃料 → 开始打造（播放动画+进度条）
    ↓ 有小游戏？
    ├─ 是 → 进入小游戏（火候/锻造/雕刻QTE）→ 小游戏得分影响品质
    └─ 否 → 等待打造完成
    ↓
打造完成 → 成功率判定
    ├─ 成功 → 生成成品（品质判定+随机词条）→ 加入背包 → 经验+
    └─ 失败 → 材料部分损失（50%~100%）→ 次品/无产出 → 少量经验
    ↓
记录打造日志 → 检查成就/任务 → 可继续打造或退出
```

#### 17.4.2 成功率计算

```gdscript
func _calculate_success_rate(recipe: CraftingRecipe, station: CraftingStation, materials: Array) -> float:
    var rate: float = recipe.base_success_rate
    var skill_level: int = _get_skill_level(recipe.category)
    var level_diff: int = skill_level - recipe.required_skill_level
    rate += clamp(level_diff * 0.03, -0.2, 0.3)
    rate += station.success_rate_bonus
    var material_quality_avg: float = _get_material_quality_average(materials)
    rate += (material_quality_avg - 1.0) * 0.1
    var optional_used: int = _count_optional_materials_used(recipe, materials)
    rate += optional_used * 0.05
    if recipe.has_minigame and minigame_active:
        rate += minigame_score * 0.15
    if _has_master_guidance_buff(recipe.category):
        rate += 0.1
    return clamp(rate, 0.05, 0.99)
```

#### 17.4.3 成品品质判定

```gdscript
func _roll_quality(recipe: CraftingRecipe, station: CraftingStation, materials: Array, success: bool) -> int:
    if not success:
        return ItemQuality.COMMON
    var quality_roll: float = randf()
    var quality_modifier: float = 0.0
    var skill_level: int = _get_skill_level(recipe.category)
    quality_modifier += (skill_level - recipe.required_skill_level) * 0.02
    quality_modifier += station.quality_bonus
    var material_quality_avg: float = _get_material_quality_average(materials)
    quality_modifier += (material_quality_avg - 1.0) * 0.15
    if recipe.has_minigame and minigame_active:
        quality_modifier += minigame_score * 0.2
    quality_modifier += GameServices.player.luck * 0.005
    var min_q: int = recipe.result_quality_min
    var max_q: int = recipe.result_quality_max
    var range: int = max_q - min_q
    if range <= 0:
        return min_q
    var thresholds: Array[float] = []
    var base_weights: Array[float] = [0.5, 0.3, 0.15, 0.04, 0.01]
    for i in range + 1:
        var idx: int = min(i, base_weights.size() - 1)
        var w: float = base_weights[idx] * (1.0 + quality_modifier * (range - i))
        thresholds.append(w)
    var total: float = 0.0
    for t in thresholds:
        total += t
    var roll: float = randf() * total
    var accumulated: float = 0.0
    for i in thresholds.size():
        accumulated += thresholds[i]
        if roll <= accumulated:
            return min_q + i
    return max_q
```

#### 17.4.4 随机词条生成

```gdscript
func _roll_random_affixes(recipe: CraftingRecipe, quality: int) -> Array[ItemAffix]:
    var result: Array[ItemAffix] = []
    if recipe.random_affix_chance <= 0 or recipe.random_affix_count <= 0:
        return result
    var affix_chance: float = recipe.random_affix_chance + (quality - ItemQuality.COMMON) * 0.1
    var affix_count: int = recipe.random_affix_count
    if quality >= ItemQuality.EPIC:
        affix_count += 1
    var affix_pool: Array[ItemAffix] = ConfigManager.get_affix_pool_for_category(recipe.category)
    var available: Array[ItemAffix] = affix_pool.duplicate()
    for i in affix_count:
        if randf() > affix_chance or available.is_empty():
            break
        var idx: int = randi() % available.size()
        var affix: ItemAffix = available[idx].duplicate()
        affix.value = randf_range(affix.value_min, affix.value_max)
        result.append(affix)
        available.remove_at(idx)
    return result
```

func attempt_research(materials: Array[Dictionary], category: int) -> ResearchResult:
    var combo_hash: String = _generate_material_hash(materials)
    if researched_combos.has(combo_hash):
        return ResearchResult.fail("已研究过此组合")
    researched_combos.append(combo_hash)
    var hidden_recipes: Array[CraftingRecipe] = ConfigManager.get_hidden_recipes(category)

            unlocked_recipes.append(recipe.id)

            EventBus.crafting_recipe_researched.emit(recipe.id)
            return ResearchResult.success(recipe.id)

    return ResearchResult.partial("未发现配方，获得少量研究经验")

```gdscript
func _gain_skill_exp(category: int, amount: int) -> void:
    var current_level: int = skill_levels.get(category, 1)
    var current_exp: int = skill_exp.get(category, 0)
    var exp_next: int = skill_exp_next.get(category, 100)
    current_exp += amount
    while current_exp >= exp_next and current_level < 100:
        current_exp -= exp_next
        current_level += 1
        exp_next = int(exp_next * 1.2 + 50)
        EventBus.crafting_skill_level_up.emit(category, current_level)
        EventBus.notification_show.emit("%s技能提升至%d级" % [_get_category_name(category), current_level])
        _check_recipe_unlock(category, current_level)
    skill_levels[category] = current_level
    skill_exp[category] = current_exp
    skill_exp_next[category] = exp_next
```

func _get_affection_level(affection: int) -> int:
    if affection >= 100: return AffectionLevel.DEVOTED
    elif affection >= 80: return AffectionLevel.LOVED
    elif affection >= 60: return AffectionLevel.CLOSE
    elif affection >= 40: return AffectionLevel.FRIENDLY
    elif affection >= 20: return AffectionLevel.ACQUAINTANCE
    else: return AffectionLevel.STRANGER

func add_affection(npc_id: String, amount: int, source: String = "") -> void:
    var current: int = affections.get(npc_id, 0)
    var old_level: int = _get_affection_level(current)
    var new_value: int = clamp(current + amount, 0, 100)
    affections[npc_id] = new_value
    var new_level: int = _get_affection_level(new_value)
    affection_levels[npc_id] = new_level
    if amount != 0:
        _add_log(npc_id, BondActionType.INTERACT, source, amount)
    if new_level > old_level:
        EventBus.bond_affection_level_up.emit(npc_id, new_level)
        _check_affection_events(npc_id, new_level)
    EventBus.bond_affection_changed.emit(npc_id, new_value, amount)

func give_gift(npc_id: String, item_instance_id: String) -> GiftResult:
    var npc_data: CharacterRelationData = ConfigManager.get_relation_data(npc_id)
    if npc_data == null:
        return GiftResult.fail("NPC不存在")
    var item: ItemInstance = GameServices.inventory.get_item_by_instance(item_instance_id)
    if item == null:
        return GiftResult.fail("物品不存在")
    var affection_gain: int = 5
    var reaction: int = GiftReaction.NEUTRAL
    if npc_data.loved_gifts.has(item.item_id) or _is_in_category(item.item_id, npc_data.loved_gifts):
        affection_gain = 20
        reaction = GiftReaction.LOVED
    elif npc_data.liked_gifts.has(item.item_id) or _is_in_category(item.item_id, npc_data.liked_gifts):
        affection_gain = 12
        reaction = GiftReaction.LIKED
    elif npc_data.disliked_gifts.has(item.item_id):
        affection_gain = -5
        reaction = GiftReaction.DISLIKED
    else:
        affection_gain = 3
        reaction = GiftReaction.NEUTRAL
    var gift_count: int = gift_count.get(npc_id, 0)
    if gift_count > 5:
        affection_gain = int(affection_gain * 0.5)
    gift_count[npc_id] = gift_count + 1

    add_affection(npc_id, affection_gain, "gift:%s" % item.item_id)
    if reaction == GiftReaction.DISLIKED:
        EventBus.bond_gift_disliked.emit(npc_id, item.item_id)
    EventBus.bond_gift_given.emit(npc_id, item.item_id, affection_gain, reaction)

    var ex_spouse: String = spouse_id
    spouse_id = ""
    spouse_equipped = false
    total_divorces += 1
    add_affection(ex_spouse, -50, "divorce")
    spouse_location = ""
    EventBus.bond_divorced.emit(ex_spouse, reason)
    return true

func break_sworn_brotherhood(npc_id: String, reason: String = "") -> bool:
    if not sworn_brothers.has(npc_id):
        return false
    sworn_brothers.erase(npc_id)
    add_affection(npc_id, -30, "break_sworn")
    EventBus.bond_sworn_broken.emit(npc_id, reason)
    return true

func try_conceive() -> ConceiveResult:
    if not BondService.is_married():
        return ConceiveResult.fail("未结婚")

        return ConceiveResult.fail("已在孕期")

        EventBus.child_difficult_labor_started.emit()
        if randf() < 0.3:
            _pregnancy.pregnancy_health = max(20, _pregnancy.pregnancy_health - 30)
    var child := ChildData.new()
    child.child_id = _generate_child_id()
    child.gender = _pregnancy.baby_gender_roll
    child.birth_date = TimeService.get_current_day()
    child.age_days = 0
    child.stage = ChildStage.INFANT
    child.father_id = _pregnancy.father_id
    child.mother_id = _pregnancy.mother_id
    child.strength = _inherit_stat(GameServices.player.strength, _get_spouse_stat("strength"))
    child.constitution = _inherit_stat(GameServices.player.constitution, _get_spouse_stat("constitution"))
    child.agility = _inherit_stat(GameServices.player.agility, _get_spouse_stat("agility"))
    child.wisdom = _inherit_stat(GameServices.player.wisdom, _get_spouse_stat("wisdom"))
    child.luck = _inherit_stat(GameServices.player.luck, _get_spouse_stat("luck"))
    child.focus = _inherit_stat(GameServices.player.focus, _get_spouse_stat("focus"))
    child.talents = _pregnancy.baby_talent_roll
    child.talent_levels = _roll_talent_levels(child.talents)
    if is_difficult:
        for talent_id in child.talents:
            child.talent_levels[talent_id] = max(1, child.talent_levels[talent_id] - 1)
    child.appearance_seed = randi()
    child.hair_color = _inherit_appearance("hair")
    child.eye_color = _inherit_appearance("eye")
    child.face_type = randi() % 10
    child.health = 80 + randi() % 20
    child.mood = 50
    child.hunger = 100
    child.affection_to_player = 30
    child.affection_to_mother = 50
    for sibling in _children:
        sibling.siblings.append(child.child_id)
        child.siblings.append(sibling.child_id)
    _children.append(child)
    _pregnancy = PregnancyState.new()
    EventBus.child_born.emit(child, is_difficult)
    EventBus.achievement_check.emit("child_born", child.gender)

#### 19.4.4 属性继承算法

```gdscript
func _inherit_stat(parent_a: int, parent_b: int) -> int:
    var base: float = (parent_a + parent_b) / 2.0
    var mutation: float = randf_range(0.8, 1.2)
    var talent_bonus: float = 1.0
    var result: int = int(base * mutation * talent_bonus * 0.2)
    return clamp(result, 1, 20)

func _roll_inherited_talents() -> Array[String]:
    var parent_talents: Array[String] = _get_parent_talents()
    var result: Array[String] = []
    var talent_count: int = 1 + randi() % 3
    for i in talent_count:
        if parent_talents.is_empty():
            result.append(_get_random_talent())
        else:
            if randf() < 0.7:
                var idx: int = randi() % parent_talents.size()
                result.append(parent_talents[idx])
                parent_talents.remove_at(idx)
            else:
                result.append(_get_random_talent())
    return result

func _roll_talent_levels(talents: Array[String]) -> Dictionary:
    var result: Dictionary = {}
    for talent_id in talents:
        var parent_level: int = _get_parent_talent_level(talent_id)
        var level: int = clamp(parent_level + randi_range(-1, 1), 1, 5)
        result[talent_id] = level
    return result
```

#### 19.4.6 孩子属性成长

```gdscript
func _grow_child_stats(child: ChildData, days_passed: int) -> void:
    var years_passed: float = days_passed / 365.0
    var stage_multiplier: float = 1.0
    match child.stage:
        ChildStage.INFANT: stage_multiplier = 0.3
        ChildStage.TODDLER: stage_multiplier = 0.8
        ChildStage.CHILD: stage_multiplier = 1.5
        ChildStage.ADULT: stage_multiplier = 0.0
    if stage_multiplier <= 0:
        return
    var education_bonus: float = 1.0 + child.education_level * 0.05
    var martial_bonus: float = 1.0 + child.martial_level * 0.05
    var talent_bonus: float = 1.0
    for talent_id in child.talents:
        if _is_physical_talent(talent_id):
            talent_bonus += child.talent_levels.get(talent_id, 1) * 0.1
    var stats: Dictionary = {
        "strength": child.strength, "constitution": child.constitution,
        "agility": child.agility, "wisdom": child.wisdom,
        "luck": child.luck, "focus": child.focus,
    }
    for stat_name in stats:
        var base_growth: float = growth_config.stat_growth_per_year.get(stat_name, 1.0)
        var growth: float = base_growth * years_passed * stage_multiplier * talent_bonus
        if stat_name == "wisdom":
            growth *= education_bonus
        elif stat_name in ["strength", "agility", "constitution"]:
            growth *= martial_bonus
        stats[stat_name] = min(100, stats[stat_name] + int(growth))
    child.strength = stats.strength
    child.constitution = stats.constitution
    child.agility = stats.agility
    child.wisdom = stats.wisdom
    child.luck = stats.luck
    child.focus = stats.focus
```

        return false

        EventBus.notification_show.emit("孩子资质不足，无法学习此武功")
        return false

    var progress: int = child.ability_progress.get(ability_id, 0)
    var learn_speed: float = 1.0 + child.wisdom * 0.02 + _get_talent_bonus(child, ability_id)

        child.ability_progress.erase(ability_id)
        child.martial_level += 1

        child.ability_progress[ability_id] = progress
    child.strength = min(100, child.strength + 1)
    child.agility = min(100, child.agility + 1)
    child.affection_to_player = min(100, child.affection_to_player + 1)
    EventBus.child_martial_trained.emit(child_id, ability_id, progress)
